  trading_rules:

    - rule: "Ignore sunk costs."
      checklist:

        - Has edge structurally degraded?

    - rule: "Use adaptive persistence."
      implementation:
        - Persist through:
            - temporary discomfort
            - variance
            - drawdowns
        - Exit when:
            - edge decays
            - regime changes
            - assumptions fail

    - rule: "Maintain exploration budget."
      implementation:
