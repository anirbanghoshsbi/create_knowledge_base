operational_rules:

  trading_rules:

    - rule: "Understand mechanisms, not just signals."
      checklist:
        - Why should this edge exist?
        - What structural forces drive it?
        - Under what regimes does it fail?
        - Is this causal or correlative?

    - rule: "Practice adaptive simulation."
      implementation:
        - Study:
            - alternative scenarios
            - regime transitions
            - volatility shocks
            - liquidity changes

    - rule: "Avoid procedural overfitting."
      checklist:
        - Am I memorizing patterns mechanically?
        - Can I explain the abstraction?
        - Does this generalize?

    - rule: "Preserve exploratory research safely."
      implementation:
        - Sandbox experimentation.
        - Separate production systems.
        - Maintain research journals.

  machine_learning_rules:

    - rule: "Optimize for transfer."
      checklist:
        - Does model generalize?
        - Are representations abstract?
        - Is training distribution too narrow?

    - rule: "Allow exploratory learning."
      implementation:
        - Use self-supervised methods.
        - Encourage representation diversity.
        - Stress-test against novel conditions.

  learning_rules:

    - rule: "Prioritize abstraction formation."
      checklist:
        - Can this transfer?
        - What invariant principle exists?
        - Can I apply this elsewhere?

    - rule: "Learn through creation and improvisation."
      implementation:
        - Build projects.
        - Simulate scenarios.
        - Generate analogies.
        - Teach concepts actively.
