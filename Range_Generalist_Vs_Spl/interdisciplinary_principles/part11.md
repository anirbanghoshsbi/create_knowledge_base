operational_rules:

  trading_rules:

    - rule: "Treat anomalies as information."
      checklist:
        - What assumption failed?
        - Is regime structure changing?
        - Is this noise or structural shift?
        - What hidden variable emerged?

    - rule: "Classify before acting."
      checklist:
        - What type of environment exists?
        - Which strategy matches this regime?
        - Is this trend, panic, rotation, or compression?

    - rule: "Do not persist blindly."
      implementation:
        - Define invalidation criteria.
        - Track statistical edge decay.
        - Stop forcing broken systems.

    - rule: "Build diversity into research."
      implementation:
        - Study:
            - psychology
            - complexity science
            - AI
            - ecology
            - macroeconomics
            - behavioral finance

  machine_learning_rules:

    - rule: "Study failure cases aggressively."
      checklist:
        - What outliers exist?
        - What examples fail consistently?
        - What assumptions broke?

    - rule: "Use diverse model perspectives."
      implementation:
        - Ensemble methods
        - Cross-regime validation
        - Heterogeneous features
        - Adversarial testing

  decision_rules:

    - rule: "Distinguish persistence from stubbornness."
      checklist:
        - Is there evidence of fit?
        - Is feedback improving?
        - Am I persisting due to identity attachment?

    - rule: "Allow nonlinear exploration."
      implementation:
        - Maintain exploration windows.
        - Preserve optionality.
        - Treat wandering as information gathering.
